# ===== SHARED VOLUME BETWEEN CONTAINERS =====
docker volume create shared_vol
docker run -it --name container1 -v shared_vol:/data ubuntu bash
# inside:
echo "Shared data from container 1" > /data/shared.txt
cat /data/shared.txt
exit

docker run -it --name container2 -v shared_vol:/data ubuntu bash
# inside:
cat /data/shared.txt
echo "Updated by container 2" >> /data/shared.txt
cat /data/shared.txt
exit

docker start container1
docker exec -it container1 cat /data/shared.txt

docker rm -f container1 container2
docker volume rm shared_vol
